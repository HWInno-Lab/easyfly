cmake_minimum_required(VERSION 3.16)
project(vision_consumer LANGUAGES CXX)
find_package(easyfly_detection_api REQUIRED)
find_package(easyfly_single_tracking_api REQUIRED)
find_package(easyfly_yolo_onnx REQUIRED)
find_package(easyfly_tag_pose REQUIRED)
find_package(easyfly_csrt REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(vision_consumer main.cpp)
target_link_libraries(vision_consumer PRIVATE easyfly_yolo_onnx::easyfly_yolo_onnx
  easyfly_tag_pose::easyfly_tag_pose easyfly_csrt::easyfly_csrt)
target_compile_options(vision_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(EASYFLY_CHECK_LEGACY)
  find_package(easyfly_object_detection REQUIRED)
  find_package(easyfly_apriltag_detection REQUIRED)
  find_package(easyfly_single_object_tracking REQUIRED)
  target_compile_definitions(vision_consumer PRIVATE EASYFLY_CHECK_LEGACY)
  target_include_directories(vision_consumer PRIVATE ${easyfly_object_detection_INCLUDE_DIRS}
    ${easyfly_apriltag_detection_INCLUDE_DIRS} ${easyfly_single_object_tracking_INCLUDE_DIRS})
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/detection_probe.cpp")
  set(_probe detection_probe.cpp)
else()
  set(_probe ../probes/detection_probe.cpp)
endif()
add_executable(detection_probe ${_probe})
target_link_libraries(detection_probe PRIVATE easyfly_yolo_onnx::easyfly_yolo_onnx yaml-cpp)
enable_testing()
add_test(NAME installed_vision_wrappers COMMAND vision_consumer)
add_executable(vision_probe vision_probe.cpp)
target_link_libraries(vision_probe PRIVATE easyfly_tag_pose::easyfly_tag_pose easyfly_csrt::easyfly_csrt)
