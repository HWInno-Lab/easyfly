cmake_minimum_required(VERSION 3.16)
project(sensor_sdk_consumer LANGUAGES C CXX)
find_package(realsense2 2.56 REQUIRED)
find_library(LIVOX_LIBRARY NAMES livox_lidar_sdk_shared REQUIRED)
find_path(LIVOX_INCLUDE NAMES livox_lidar_api.h REQUIRED)
add_executable(sdk_consumer sdk.cpp)
target_compile_features(sdk_consumer PRIVATE cxx_std_17)
target_compile_options(sdk_consumer PRIVATE -Wall -Wextra)
target_include_directories(sdk_consumer PRIVATE ${LIVOX_INCLUDE})
target_link_libraries(sdk_consumer PRIVATE ${LIVOX_LIBRARY} realsense2::realsense2)
option(BUILD_DRIVER_CONSUMER "Load installed ROS component libraries without creating devices" OFF)
if(BUILD_DRIVER_CONSUMER)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rclcpp_components REQUIRED)
  find_package(class_loader REQUIRED)
  find_package(livox_ros_driver2 REQUIRED)
  find_package(realsense2_camera_msgs REQUIRED)
  add_executable(driver_consumer drivers.cpp)
  target_compile_features(driver_consumer PRIVATE cxx_std_17)
  target_compile_options(driver_consumer PRIVATE -Wall -Wextra)
  ament_target_dependencies(driver_consumer rclcpp rclcpp_components class_loader
    livox_ros_driver2 realsense2_camera_msgs)
endif()
