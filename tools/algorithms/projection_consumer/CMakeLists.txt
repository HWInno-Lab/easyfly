cmake_minimum_required(VERSION 3.16)
project(projection_consumer LANGUAGES CXX)
find_package(easyfly_projection REQUIRED)
add_executable(projection_consumer main.cpp)
target_link_libraries(projection_consumer PRIVATE easyfly_projection::easyfly_projection)
target_compile_options(projection_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
set(_probe ../probes/projection_probe.cpp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/projection_probe.cpp")
  set(_probe projection_probe.cpp)
endif()
add_executable(projection_probe ${_probe})
target_link_libraries(projection_probe PRIVATE easyfly_projection::easyfly_projection)
enable_testing()
add_test(NAME installed_projection_geometry COMMAND projection_consumer)
