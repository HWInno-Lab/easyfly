cmake_minimum_required(VERSION 3.16)
project(ground_consumer LANGUAGES C CXX)
find_package(easyfly_ground_segmentation REQUIRED)
add_executable(ground_consumer main.cpp)
target_link_libraries(ground_consumer PRIVATE easyfly_ground_segmentation::easyfly_ground_segmentation)
target_compile_options(ground_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(EASYFLY_CHECK_LEGACY)
  find_package(easyfly_semantic_map REQUIRED)
  target_compile_definitions(ground_consumer PRIVATE EASYFLY_CHECK_LEGACY)
  target_include_directories(ground_consumer PRIVATE ${easyfly_semantic_map_INCLUDE_DIRS})
endif()
set(_probe ../probes/ground_probe.cpp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ground_probe.cpp")
  set(_probe ground_probe.cpp)
endif()
add_executable(ground_probe ${_probe})
target_link_libraries(ground_probe PRIVATE easyfly_ground_segmentation::easyfly_ground_segmentation)
enable_testing()
add_test(NAME installed_ground_computation COMMAND ground_consumer)
