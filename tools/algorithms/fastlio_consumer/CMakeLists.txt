cmake_minimum_required(VERSION 3.16)
project(fastlio_message_consumer LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(fast_lio REQUIRED)
find_package(rclcpp REQUIRED)
add_executable(fastlio_message_consumer main.cpp)
target_compile_features(fastlio_message_consumer PRIVATE cxx_std_17)
ament_target_dependencies(fastlio_message_consumer fast_lio rclcpp)
target_compile_options(fastlio_message_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME installed_pose6d_roundtrip COMMAND fastlio_message_consumer)
