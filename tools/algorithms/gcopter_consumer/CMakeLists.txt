cmake_minimum_required(VERSION 3.16)
project(gcopter_installed_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(easyfly_trajectory_optimizer REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer easyfly_trajectory_optimizer::trajectory_core)
find_package(easyfly_astar REQUIRED)
add_executable(compare compare.cpp)
target_link_libraries(compare easyfly_trajectory_optimizer::trajectory_core easyfly_astar::easyfly_astar)
target_compile_options(compare PRIVATE -O2)
