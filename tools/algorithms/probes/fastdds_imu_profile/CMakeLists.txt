cmake_minimum_required(VERSION 3.16)
project(fastdds_imu_profile_probe)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rmw_fastrtps_shared_cpp REQUIRED)
find_package(fastrtps REQUIRED)
add_executable(verify_profile verify_profile.cpp)
target_compile_features(verify_profile PRIVATE cxx_std_17)
ament_target_dependencies(verify_profile rclcpp sensor_msgs rmw_fastrtps_shared_cpp)
target_link_libraries(verify_profile fastrtps)
add_executable(verify_local_transport verify_local_transport.cpp)
target_compile_features(verify_local_transport PRIVATE cxx_std_17)
ament_target_dependencies(verify_local_transport rclcpp sensor_msgs rmw_fastrtps_shared_cpp)
target_link_libraries(verify_local_transport fastrtps)
add_executable(imu_receipts imu_receipts.cpp)
target_compile_features(imu_receipts PRIVATE cxx_std_17)
ament_target_dependencies(imu_receipts rclcpp sensor_msgs)

option(BUILD_GAZEBO_CAMERA_PROBE "Build the optional native Gazebo image observer" OFF)
if(BUILD_GAZEBO_CAMERA_PROBE)
  find_package(gazebo_dev REQUIRED)
  find_package(OpenSSL REQUIRED)
  add_executable(gazebo_camera_receipts gazebo_camera_receipts.cpp)
  target_compile_features(gazebo_camera_receipts PRIVATE cxx_std_17)
  ament_target_dependencies(gazebo_camera_receipts gazebo_dev)
  target_link_libraries(gazebo_camera_receipts OpenSSL::Crypto)
endif()
