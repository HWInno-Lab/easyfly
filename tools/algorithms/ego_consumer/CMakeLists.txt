cmake_minimum_required(VERSION 3.16)
project(ego_consumer LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(bspline_opt REQUIRED)
find_package(path_searching REQUIRED)
find_package(plan_env REQUIRED)
find_package(traj_utils_so REQUIRED)
find_package(traj_utils REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(rclcpp REQUIRED)
add_executable(ego_consumer main.cpp)
target_compile_features(ego_consumer PRIVATE cxx_std_17)
ament_target_dependencies(ego_consumer bspline_opt path_searching plan_env traj_utils_so traj_utils quadrotor_msgs rclcpp)
target_compile_options(ego_consumer PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME installed_spline_and_messages COMMAND ego_consumer)
