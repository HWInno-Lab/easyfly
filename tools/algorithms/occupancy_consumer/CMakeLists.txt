cmake_minimum_required(VERSION 3.16)
project(occupancy_consumer LANGUAGES CXX)
find_package(easyfly_occupancy_mapping REQUIRED)
set(_probe ../probes/occupancy_probe.cpp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/occupancy_probe.cpp")
  set(_probe occupancy_probe.cpp)
endif()
add_executable(occupancy_probe ${_probe})
target_link_libraries(occupancy_probe PRIVATE easyfly_occupancy_mapping::easyfly_occupancy_mapping)
target_compile_options(occupancy_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(EASYFLY_CHECK_LEGACY)
  # ROS package metadata imports the PCL ground library, whose config requires C/MPI.
  enable_language(C)
  find_package(easyfly_semantic_map REQUIRED)
  target_compile_definitions(occupancy_probe PRIVATE EASYFLY_CHECK_LEGACY)
  target_include_directories(occupancy_probe PRIVATE ${easyfly_semantic_map_INCLUDE_DIRS})
endif()
enable_testing()
add_test(NAME installed_occupancy_computation COMMAND occupancy_probe)
