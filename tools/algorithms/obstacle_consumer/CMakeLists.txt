cmake_minimum_required(VERSION 3.16)
project(obstacle_consumer LANGUAGES C CXX)
find_package(easyfly_obstacle_geometry REQUIRED)
set(_probe ../probes/obstacle_probe.cpp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/obstacle_probe.cpp")
  set(_probe obstacle_probe.cpp)
endif()
add_executable(obstacle_probe ${_probe})
target_link_libraries(obstacle_probe PRIVATE easyfly_obstacle_geometry::easyfly_obstacle_geometry)
target_compile_options(obstacle_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(EASYFLY_CHECK_LEGACY)
  find_package(easyfly_semantic_map REQUIRED)
  target_compile_definitions(obstacle_probe PRIVATE EASYFLY_CHECK_LEGACY)
  target_include_directories(obstacle_probe PRIVATE ${easyfly_semantic_map_INCLUDE_DIRS})
endif()
enable_testing()
add_test(NAME installed_obstacle_computation COMMAND obstacle_probe)
