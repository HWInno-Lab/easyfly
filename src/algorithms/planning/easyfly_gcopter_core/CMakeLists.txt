cmake_minimum_required(VERSION 3.8)
project(easyfly_gcopter_core)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
set(vendor "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/algorithms/planning/gcopter")
add_library(gcopter INTERFACE)
target_compile_features(gcopter INTERFACE cxx_std_17)
target_include_directories(gcopter INTERFACE
  $<BUILD_INTERFACE:${vendor}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(gcopter INTERFACE Eigen3::Eigen)
install(TARGETS gcopter EXPORT gcopterTargets)
install(DIRECTORY "${vendor}/include/" DESTINATION include)
install(FILES "${vendor}/LICENSE" "${vendor}/README.md" DESTINATION share/${PROJECT_NAME})
ament_export_targets(gcopterTargets)
ament_export_dependencies(Eigen3)
ament_package()
