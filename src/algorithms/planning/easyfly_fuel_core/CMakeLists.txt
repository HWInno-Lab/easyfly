cmake_minimum_required(VERSION 3.16)
project(easyfly_fuel_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(easyfly_fuel_native REQUIRED)
add_library(fuel_core SHARED src/core.cpp)
target_include_directories(fuel_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fuel_core PUBLIC easyfly_fuel_native::fuel_native)
target_link_options(fuel_core PRIVATE -Wl,--no-undefined)
install(TARGETS fuel_core EXPORT coreTargets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(coreTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(easyfly_fuel_native)
ament_package()
