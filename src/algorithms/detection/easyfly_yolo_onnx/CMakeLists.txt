cmake_minimum_required(VERSION 3.16)
project(easyfly_yolo_onnx)
find_package(ament_cmake REQUIRED)
find_package(easyfly_detection_api REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
set(ORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/runtimes/onnxruntime" CACHE PATH "ONNX Runtime SDK root")
if(NOT EXISTS "${ORT_ROOT}/include/onnxruntime_cxx_api.h" OR NOT EXISTS "${ORT_ROOT}/lib/libonnxruntime.so")
  message(FATAL_ERROR "ONNX Runtime SDK not found at ORT_ROOT=${ORT_ROOT}")
endif()
get_filename_component(_ort_library "${ORT_ROOT}/lib/libonnxruntime.so" REALPATH)
get_filename_component(_ort_filename "${_ort_library}" NAME)
execute_process(COMMAND "${CMAKE_READELF}" -d "${_ort_library}" OUTPUT_VARIABLE _ort_dynamic RESULT_VARIABLE _ort_readelf_status)
string(REGEX MATCH "SONAME[^\n]*\\[([^]]+)\\]" _ort_soname_match "${_ort_dynamic}")
set(_ort_soname "${CMAKE_MATCH_1}")
if(NOT _ort_readelf_status EQUAL 0 OR NOT _ort_soname)
  message(FATAL_ERROR "Cannot determine ONNX Runtime SONAME: ${_ort_library}")
endif()
add_library(${PROJECT_NAME} SHARED src/onnx_backend.cpp)
add_library(${PROJECT_NAME}::${PROJECT_NAME} ALIAS ${PROJECT_NAME})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(${PROJECT_NAME} PRIVATE "${ORT_ROOT}/include")
target_link_libraries(${PROJECT_NAME} PUBLIC easyfly_detection_api::easyfly_detection_api PRIVATE "${_ort_library}")
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra)
set_target_properties(${PROJECT_NAME} PROPERTIES BUILD_RPATH "${ORT_ROOT}/lib" INSTALL_RPATH "$ORIGIN")
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(FILES "${_ort_library}" DESTINATION lib)
file(GLOB _ort_providers "${ORT_ROOT}/lib/libonnxruntime_providers_*.so")
if(_ort_providers)
  install(FILES ${_ort_providers} DESTINATION lib)
endif()
configure_file(cmake/install_runtime.cmake.in install_runtime.cmake @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/install_runtime.cmake")
foreach(_notice LICENSE ThirdPartyNotices.txt VERSION_NUMBER GIT_COMMIT_ID)
  if(EXISTS "${ORT_ROOT}/${_notice}")
    install(FILES "${ORT_ROOT}/${_notice}" DESTINATION share/${PROJECT_NAME}/onnxruntime)
  endif()
endforeach()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_postprocess test/test_postprocess.cpp)
  target_link_libraries(test_postprocess ${PROJECT_NAME})
endif()
ament_package(CONFIG_EXTRAS cmake/dependencies.cmake)
