cmake_minimum_required(VERSION 3.8)
project(easyfly_rviz_plugins)
find_package(ament_cmake REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rviz_common 11.2.26 EXACT REQUIRED)
find_package(rviz_rendering 11.2.26 EXACT REQUIRED)
find_package(rviz_default_plugins 11.2.26 EXACT REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(easyfly_rviz_odometry SHARED src/odometry_display.cpp)
set_target_properties(easyfly_rviz_odometry PROPERTIES EXPORT_NAME odometry_display)
target_compile_features(easyfly_rviz_odometry PUBLIC cxx_std_17)
target_include_directories(easyfly_rviz_odometry PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(easyfly_rviz_odometry PUBLIC pluginlib)
target_link_libraries(easyfly_rviz_odometry PUBLIC rviz_common::rviz_common
  rviz_rendering::rviz_rendering rviz_default_plugins::rviz_default_plugins Qt5::Widgets)
target_compile_options(easyfly_rviz_odometry PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS easyfly_rviz_odometry EXPORT odometryTargets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(odometryTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(pluginlib rviz_common rviz_rendering rviz_default_plugins)
if(BUILD_TESTING)
  find_package(ament_lint_auto REQUIRED)
  ament_lint_auto_find_test_dependencies()
endif()
ament_package(CONFIG_EXTRAS "cmake/dependencies.cmake")
