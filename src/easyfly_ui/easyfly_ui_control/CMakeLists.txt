cmake_minimum_required(VERSION 3.8)
project(easyfly_ui_control)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(easyfly_msgs REQUIRED)
find_package(easyfly_mission_format REQUIRED)
find_package(jsoncpp REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(${PROJECT_NAME} SHARED src/journal.cpp src/controller.cpp src/parameters.cpp
  src/import_journal.cpp src/importer.cpp)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} PUBLIC rclcpp easyfly_msgs)
target_link_libraries(${PROJECT_NAME} PRIVATE JsonCpp::JsonCpp
  easyfly_mission_format::easyfly_mission_format)
install(TARGETS ${PROJECT_NAME} EXPORT controlTargets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(controlTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp easyfly_msgs easyfly_mission_format)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_importer test/test_importer.cpp TIMEOUT 60)
  target_include_directories(test_importer PRIVATE src)
  target_link_libraries(test_importer ${PROJECT_NAME} easyfly_mission_format::easyfly_mission_format)
  set_tests_properties(test_importer PROPERTIES
    ENVIRONMENT "ROS_DOMAIN_ID=128;ROS_LOCALHOST_ONLY=1")
  ament_add_gtest(test_parameters test/test_parameters.cpp TIMEOUT 60)
  target_link_libraries(test_parameters ${PROJECT_NAME})
  set_tests_properties(test_parameters PROPERTIES
    ENVIRONMENT "ROS_DOMAIN_ID=123;ROS_LOCALHOST_ONLY=1")
  ament_add_gtest(test_journal test/test_journal.cpp)
  target_include_directories(test_journal PRIVATE src)
  target_link_libraries(test_journal ${PROJECT_NAME})
  ament_add_gtest(test_controller test/test_controller.cpp TIMEOUT 90)
  target_include_directories(test_controller PRIVATE src)
  target_link_libraries(test_controller ${PROJECT_NAME})
  target_link_libraries(test_controller easyfly_mission_format::easyfly_mission_format)
  set_tests_properties(test_controller PROPERTIES
    ENVIRONMENT "ROS_DOMAIN_ID=113;ROS_LOCALHOST_ONLY=1")
  find_package(ament_lint_auto REQUIRED)
  ament_lint_auto_find_test_dependencies()
endif()
ament_package()
