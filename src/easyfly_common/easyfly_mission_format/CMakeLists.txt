cmake_minimum_required(VERSION 3.8)
project(easyfly_mission_format)

find_package(ament_cmake REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenSSL REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(${PROJECT_NAME} SHARED src/mission_format.cpp src/mission_metadata.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} PRIVATE yaml-cpp OpenSSL::Crypto)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mission_validate src/mission_validate.cpp)
target_link_libraries(mission_validate ${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} EXPORT mission_formatTargets LIBRARY DESTINATION lib)
install(TARGETS mission_validate DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(mission_formatTargets HAS_LIBRARY_TARGET)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_legacy_mission test/test_legacy_mission.cpp)
  target_link_libraries(test_legacy_mission ${PROJECT_NAME})
  ament_add_gtest(test_mission_format test/test_mission_format.cpp)
  target_link_libraries(test_mission_format ${PROJECT_NAME})
  ament_add_gtest(test_metadata test/test_metadata.cpp)
  target_link_libraries(test_metadata ${PROJECT_NAME})
  find_package(ament_lint_auto REQUIRED)
  ament_lint_auto_find_test_dependencies()
endif()
ament_package()
