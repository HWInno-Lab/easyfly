cmake_minimum_required(VERSION 3.8)
project(easyfly_apriltag_detection)
find_package(ament_cmake REQUIRED)
find_package(easyfly_tag_pose REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(easyfly_msgs REQUIRED)
find_package(OpenCV REQUIRED)
add_executable(apriltag_detection_node src/tag_node.cpp)
target_compile_features(apriltag_detection_node PRIVATE cxx_std_17)
target_include_directories(apriltag_detection_node PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(apriltag_detection_node rclcpp sensor_msgs easyfly_msgs)
target_link_libraries(apriltag_detection_node ${OpenCV_LIBRARIES} easyfly_tag_pose::easyfly_tag_pose)
install(TARGETS apriltag_detection_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_lint_auto REQUIRED)
  ament_lint_auto_find_test_dependencies()
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_clock_ros test/test_clock_ros.py TIMEOUT 30
    ENV ROS_DOMAIN_ID=105 ROS_LOG_DIR=/tmp/easyfly_d4_clock_105)
  ament_add_pytest_test(test_tag_ros test/test_tag_ros.py TIMEOUT 60
    ENV ROS_DOMAIN_ID=105 ROS_LOG_DIR=/tmp/easyfly_d4_tag_ros)
  ament_add_pytest_test(test_duplicate_ros test/test_duplicate_ros.py TIMEOUT 60
    ENV ROS_DOMAIN_ID=106 ROS_LOG_DIR=/tmp/easyfly_d5_tag_duplicate)
endif()
add_executable(tag_fixture test/tag_fixture.cpp)
target_include_directories(tag_fixture PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(tag_fixture ${OpenCV_LIBRARIES})
install(TARGETS tag_fixture DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include/${PROJECT_NAME})
ament_export_include_directories(include/${PROJECT_NAME})
ament_export_dependencies(easyfly_tag_pose)
ament_package()
