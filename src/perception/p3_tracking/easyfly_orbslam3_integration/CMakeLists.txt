cmake_minimum_required(VERSION 3.16)
project(easyfly_orbslam3_integration)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(easyfly_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(easyfly_orbslam3_core REQUIRED)
add_executable(orbslam3_rgbd src/rgbd_node.cpp)
target_include_directories(orbslam3_rgbd PRIVATE include)
target_compile_features(orbslam3_rgbd PRIVATE cxx_std_17)
ament_target_dependencies(orbslam3_rgbd rclcpp std_srvs sensor_msgs geometry_msgs diagnostic_msgs easyfly_msgs cv_bridge)
target_link_libraries(orbslam3_rgbd easyfly_orbslam3_core::easyfly_orbslam3)
install(TARGETS orbslam3_rgbd DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_inertial_contract test/test_inertial_contract.cpp)
  target_compile_features(test_inertial_contract PRIVATE cxx_std_17)
  target_include_directories(test_inertial_contract PRIVATE include)
  target_link_libraries(test_inertial_contract easyfly_orbslam3_core::easyfly_orbslam3)
  ament_add_gtest(test_pose_continuity test/test_pose_continuity.cpp)
  target_compile_features(test_pose_continuity PRIVATE cxx_std_17)
  target_include_directories(test_pose_continuity PRIVATE include)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_orbslam3_launch test/test_orbslam3_launch.py)
endif()
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
