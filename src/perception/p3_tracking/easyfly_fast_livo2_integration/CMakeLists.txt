cmake_minimum_required(VERSION 3.16)
project(easyfly_fast_livo2_integration)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/fastlivo_guard DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
find_package(ament_cmake_pytest REQUIRED)
ament_add_pytest_test(test_guard test/test_guard.py)
ament_add_pytest_test(test_launch test/test_launch.py)
endif()
ament_package()
