cmake_minimum_required(VERSION 3.8)
project(easyfly_trajectory_optimizer)

find_package(ament_cmake REQUIRED)
find_package(easyfly_gcopter_core REQUIRED)
find_package(rclcpp REQUIRED)
find_package(easyfly_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(ament_index_cpp REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(trajectory_core SHARED src/plan.cpp)
target_include_directories(trajectory_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(trajectory_core PUBLIC easyfly_gcopter_core::gcopter)
target_compile_options(trajectory_core PRIVATE -O2 -Wall -Wextra)
add_executable(gcopter_worker src/worker.cpp)
target_link_libraries(gcopter_worker trajectory_core)
add_executable(gcopter_voxel_map src/voxel_map_node.cpp)
target_include_directories(gcopter_voxel_map PRIVATE include)
target_link_libraries(gcopter_voxel_map easyfly_gcopter_core::gcopter)
ament_target_dependencies(gcopter_voxel_map rclcpp easyfly_msgs sensor_msgs tf2_ros tf2_eigen)
add_executable(gcopter_navigation src/navigation_node.cpp)
target_link_libraries(gcopter_navigation trajectory_core)
ament_target_dependencies(gcopter_navigation rclcpp easyfly_msgs quadrotor_msgs ament_index_cpp)
target_compile_options(gcopter_voxel_map PRIVATE -O2 -Wall -Wextra)
target_compile_options(gcopter_navigation PRIVATE -O2 -Wall -Wextra)
install(TARGETS trajectory_core EXPORT trajectoryTargets LIBRARY DESTINATION lib)
install(TARGETS gcopter_worker gcopter_navigation gcopter_voxel_map DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_observation_grid test/test_observation_grid.cpp)
  target_link_libraries(test_observation_grid trajectory_core)
  ament_add_gtest(test_gcopter_core test/test_core.cpp)
  target_link_libraries(test_gcopter_core trajectory_core)
  target_compile_definitions(test_gcopter_core PRIVATE
    TRACKING_REQUEST_FROM_TEST="${CMAKE_CURRENT_SOURCE_DIR}/test/data/fuel_room_tracking.request")
  ament_add_pytest_test(test_gcopter_recorded_worker test/test_recorded_worker.py TIMEOUT 10)
  ament_add_pytest_test(test_gcopter_navigation_ros test/test_navigation_ros.py
    TIMEOUT 90 ENV ROS_DOMAIN_ID=197)
  ament_add_pytest_test(test_gcopter_voxel_ros test/test_voxel_ros.py
    TIMEOUT 65 ENV ROS_DOMAIN_ID=198)
endif()
ament_export_targets(trajectoryTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(easyfly_gcopter_core)
ament_package()
