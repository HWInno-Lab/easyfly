cmake_minimum_required(VERSION 3.16)
project(easyfly_fuel_exploration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(easyfly_fuel_core REQUIRED)
find_package(easyfly_trajectory_optimizer REQUIRED)
find_package(easyfly_fuel_native REQUIRED)
find_package(NLopt CONFIG REQUIRED)
add_executable(fuel_bspline_worker src/bspline_worker.cpp)
target_link_libraries(fuel_bspline_worker PRIVATE easyfly_trajectory_optimizer::trajectory_core easyfly_fuel_native::fuel_native)
target_compile_options(fuel_bspline_worker PRIVATE -O2 -Wall -Wextra)
if(TARGET NLopt::nlopt)
 target_link_libraries(fuel_bspline_worker PRIVATE NLopt::nlopt)
else()
 target_link_libraries(fuel_bspline_worker PRIVATE NLopt::nlopt_cxx)
endif()
install(TARGETS fuel_bspline_worker DESTINATION lib/${PROJECT_NAME})
add_executable(fuel_worker src/worker.cpp)
target_link_libraries(fuel_worker PRIVATE easyfly_fuel_core::fuel_core easyfly_trajectory_optimizer::trajectory_core)
target_link_options(fuel_worker PRIVATE -Wl,--no-undefined)
install(TARGETS fuel_worker DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/fuel_exploration DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_pytest REQUIRED)
 ament_add_pytest_test(test_shutdown test/test_shutdown.py TIMEOUT 20 ENV ROS_DOMAIN_ID=229 ROS_LOCALHOST_ONLY=1)
 ament_add_pytest_test(test_bspline_worker test/test_bspline_worker.py TIMEOUT 45)
 ament_add_pytest_test(test_camera test/test_camera.py TIMEOUT 30)
 ament_add_pytest_test(test_fuel_policy test/test_policy.py TIMEOUT 30)
 ament_add_pytest_test(test_fuel_clock test/test_lifecycle_clock.py TIMEOUT 30 ENV ROS_DOMAIN_ID=228 ROS_LOCALHOST_ONLY=1)
endif()
ament_package()
