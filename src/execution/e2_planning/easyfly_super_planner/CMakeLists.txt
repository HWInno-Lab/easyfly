cmake_minimum_required(VERSION 3.16)
project(easyfly_super_planner)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(super_planner REQUIRED)
find_package(easyfly_trajectory_optimizer REQUIRED)
find_package(easyfly_gcopter_core REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(easyfly_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
add_library(super_validation SHARED src/plan.cpp)
target_include_directories(super_validation PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> $<TARGET_PROPERTY:easyfly_trajectory_optimizer::trajectory_core,INTERFACE_INCLUDE_DIRECTORIES>)
target_link_libraries(super_validation PUBLIC super_planner::easyfly_super_planner)
add_library(super_execution_core SHARED src/optimize.cpp)
target_link_libraries(super_execution_core PUBLIC super_validation super_planner::easyfly_super_planner)
ament_target_dependencies(super_execution_core PUBLIC ament_index_cpp)
add_executable(super_worker src/worker.cpp)
target_link_libraries(super_worker super_execution_core)
add_executable(super_navigation src/navigation_node.cpp)
target_link_libraries(super_navigation super_validation)
ament_target_dependencies(super_navigation rclcpp easyfly_msgs quadrotor_msgs ament_index_cpp)
foreach(target super_validation super_execution_core super_worker super_navigation)
  target_compile_options(${target} PRIVATE -O2 -Wall -Wextra)
endforeach()
install(TARGETS super_validation super_execution_core EXPORT superTargets LIBRARY DESTINATION lib)
install(TARGETS super_worker super_navigation DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(superTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(super_planner easyfly_trajectory_optimizer easyfly_gcopter_core ament_index_cpp)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_super_septic test/test_septic.cpp)
  target_link_libraries(test_super_septic super_validation)
  ament_add_pytest_test(test_super_navigation_ros test/test_navigation_ros.py TIMEOUT 120 ENV ROS_DOMAIN_ID=217)
  ament_add_pytest_test(test_native_map_change test/test_native_map_change.py TIMEOUT 60
    ENV "SUPER_WORKER=$<TARGET_FILE:super_worker>")
endif()
ament_package()
