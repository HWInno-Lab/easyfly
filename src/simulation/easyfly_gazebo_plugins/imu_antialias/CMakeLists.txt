find_package(gazebo 11.10.2 EXACT REQUIRED)
add_library(easyfly_imu_antialias SHARED imu_antialias.cpp)
target_compile_features(easyfly_imu_antialias PRIVATE cxx_std_17)
target_include_directories(easyfly_imu_antialias PRIVATE include ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(easyfly_imu_antialias PRIVATE ${GAZEBO_LIBRARIES})
install(TARGETS easyfly_imu_antialias LIBRARY DESTINATION lib)
install(FILES README.md DESTINATION share/${PROJECT_NAME}/imu_antialias)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_imu_antialias test/test_window.cpp TIMEOUT 15)
  target_include_directories(test_imu_antialias PRIVATE include)
  target_compile_features(test_imu_antialias PRIVATE cxx_std_17)
  ament_add_gtest(test_imu_message_type test/test_message_type.cpp TIMEOUT 15)
  target_include_directories(test_imu_message_type PRIVATE include ${GAZEBO_INCLUDE_DIRS})
  target_compile_features(test_imu_message_type PRIVATE cxx_std_17)
  target_link_libraries(test_imu_message_type ${GAZEBO_LIBRARIES})
endif()
