find_package(gazebo 11.10.2 EXACT REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(gazebo_plugins REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(easyfly_gazebo_ros_imu SHARED src/safe_imu.cpp)
target_compile_features(easyfly_gazebo_ros_imu PRIVATE cxx_std_17)
target_include_directories(easyfly_gazebo_ros_imu PRIVATE include ${GAZEBO_INCLUDE_DIRS})
ament_target_dependencies(easyfly_gazebo_ros_imu gazebo_ros gazebo_plugins sensor_msgs)
target_link_libraries(easyfly_gazebo_ros_imu ${GAZEBO_LIBRARIES})
install(TARGETS easyfly_gazebo_ros_imu LIBRARY DESTINATION lib)
install(FILES README.md NOTICE LICENSE upstream/SHA256SUMS DESTINATION share/${PROJECT_NAME}/safe_imu)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_safe_imu_source test/test_source.py)
endif()
