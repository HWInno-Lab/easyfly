find_package(gazebo 11.10.2 EXACT REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(gazebo_plugins REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(Threads REQUIRED)
add_library(easyfly_gazebo_ros_imu SHARED src/safe_imu.cpp)
target_compile_features(easyfly_gazebo_ros_imu PRIVATE cxx_std_17)
target_include_directories(easyfly_gazebo_ros_imu PRIVATE include ${GAZEBO_INCLUDE_DIRS})
ament_target_dependencies(easyfly_gazebo_ros_imu gazebo_ros gazebo_plugins sensor_msgs)
target_link_libraries(easyfly_gazebo_ros_imu ${GAZEBO_LIBRARIES} Threads::Threads)
install(TARGETS easyfly_gazebo_ros_imu LIBRARY DESTINATION lib)
install(FILES README.md NOTICE LICENSE upstream/SHA256SUMS DESTINATION share/${PROJECT_NAME}/safe_imu)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(rcl REQUIRED)
  add_library(test_imu_publish_stall SHARED test/publish_stall.cpp)
  target_compile_features(test_imu_publish_stall PRIVATE cxx_std_17)
  ament_target_dependencies(test_imu_publish_stall rcl sensor_msgs)
  target_link_libraries(test_imu_publish_stall ${CMAKE_DL_LIBS} Threads::Threads)
  foreach(mode short overflow)
    set(stall_args)
    if(mode STREQUAL "overflow")
      list(APPEND stall_args --overflow)
    endif()
    add_test(NAME test_imu_publish_${mode} COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/test/test_publish_queue.py
      $<TARGET_FILE:easyfly_gzserver> $<TARGET_FILE:easyfly_gazebo_ros_imu>
      $<TARGET_FILE:test_imu_publish_stall> ${stall_args})
    set_tests_properties(test_imu_publish_${mode} PROPERTIES TIMEOUT 50 RUN_SERIAL TRUE)
  endforeach()
  ament_add_pytest_test(test_safe_imu_source test/test_source.py)
endif()
